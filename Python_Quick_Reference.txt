## DATA STRUCTURES - STACK (Day 13)

# Stack (LIFO - Last In, First Out)
class Stack:
    def __init__(self):
        self.items = []
    
    def push(self, item):
        """Add to top"""
        self.items.append(item)
    
    def pop(self):
        """Remove from top"""
        if not self.is_empty():
            return self.items.pop()
        return None
    
    def peek(self):
        """Look at top"""
        if not self.is_empty():
            return self.items[-1]
        return None
    
    def is_empty(self):
        return len(self.items) == 0
    
    def size(self):
        return len(self.items)

# Stack Usage
stack = Stack()
stack.push(1)
stack.push(2)
top = stack.pop()    # 2 (last in, first out)
peek = stack.peek()  # 1

# Stack Applications
# 1. Reverse string
def reverse(text):
    stack = Stack()
    for char in text:
        stack.push(char)
    result = ""
    while not stack.is_empty():
        result += stack.pop()
    return result

# 2. Undo functionality
class Editor:
    def __init__(self):
        self.history = Stack()
    
    def save_state(self, state):
        self.history.push(state)
    
    def undo(self):
        return self.history.pop()

# Queue (FIFO - First In, First Out)
class Queue:
    def __init__(self):
        self.items = []
    
    def enqueue(self, item):
        """Add to rear"""
        self.items.append(item)
    
    def dequeue(self):
        """Remove from front"""
        if not self.is_empty():
            return self.items.pop(0)
        return None
    
    def front(self):
        """Look at front"""
        if not self.is_empty():
            return self.items[0]
        return None
    
    def is_empty(self):
        return len(self.items) == 0
    
    def size(self):
        return len(self.items)

# Queue Usage
queue = Queue()
queue.enqueue(1)
queue.enqueue(2)
front = queue.dequeue()  # 1 (first in, first out)
peek = queue.front()     # 2

# Queue Applications
# 1. Customer service
class Service:
    def __init__(self):
        self.customers = Queue()
    
    def add_customer(self, name):
        self.customers.enqueue(name)
    
    def serve_next(self):
        return self.customers.dequeue()

# 2. Print queue
class Printer:
    def __init__(self):
        self.queue = Queue()
    
    def add_job(self, doc):
        self.queue.enqueue(doc)
    
    def print_next(self):

## LINKED LISTS (Day 14)

# Singly Linked List Node
class Node:
    def __init__(self, data):
        self.data = data    # Value
        self.next = None    # Pointer to next

# Singly Linked List
class LinkedList:
    def __init__(self):
        self.head = None
    
    # Append (add to end)
    def append(self, data):
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
            return
        current = self.head
        while current.next:
            current = current.next
        current.next = new_node
    
    # Prepend (add to beginning)
    def prepend(self, data):
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node
    
    # Delete
    def delete(self, data):
        if self.head and self.head.data == data:
            self.head = self.head.next
            return True
        current = self.head
        while current and current.next:
            if current.next.data == data:
                current.next = current.next.next
                return True
            current = current.next
        return False
    
    # Search
    def search(self, data):
        current = self.head
        while current:
            if current.data == data:
                return True
            current = current.next
        return False

# Traversal Pattern
current = self.head
while current:
    # Process current
    current = current.next

# Doubly Linked List Node
class DNode:
    def __init__(self, data):
        self.data = data
        self.next = None
        self.prev = None

# Doubly Linked List
class DoublyLinkedList:
    def __init__(self):
        self.head = None
        self.tail = None
    
    # Append
    def append(self, data):
        new_node = DNode(data)
        if self.head is None:
            self.head = new_node
            self.tail = new_node
            return
        self.tail.next = new_node
        new_node.prev = self.tail
        self.tail = new_node
    
    # Delete (3 cases: head, tail, middle)
    def delete(self, data):
        current = self.head
        while current:
            if current.data == data:
                if current.prev:
                    current.prev.next = current.next
                else:  # Delete head
                    self.head = current.next
                if current.next:
                    current.next.prev = current.prev
                else:  # Delete tail
                    self.tail = current.prev
                return True
            current = current.next
        return False

# When to Use Linked Lists

Use Linked Lists:
- Frequent insertions/deletions
- Unknown size
- Don't need random access
- Dynamic growth

Use Arrays:
- Need fast index access
- Known size
- Few modifications
- Cache performance matters
